-- Fast Punch
local function autoFastPunch()
    local player = game:GetService("Players").LocalPlayer
    while _G.fastPunch do
        local punchTool = player.Backpack:FindFirstChild("Punch")
        if punchTool then
            punchTool.Parent = player.Character
            task.wait(0.1)
        end

        local character = player.Character
        local punch = character and character:FindFirstChild("Punch")
        if punch and punch:FindFirstChild("attackTime") then
            punch.attackTime.Value = 0
        end

        player.muscleEvent:FireServer("punch", "leftHand")
        player.muscleEvent:FireServer("punch", "rightHand")
        task.wait(0.001)
    end

    local punch = player.Character and player.Character:FindFirstChild("Punch")
    if punch and punch:FindFirstChild("attackTime") then
        punch.attackTime.Value = 0.35
    end

    if player.Character:FindFirstChild("Punch") then
        player.Character.Punch.Parent = player.Backpack
    end
end

-- Auto Punch
local function autoPunch()
    local player = game.Players.LocalPlayer
    local playerName = player.Name
    _G.punchanim = true

    game:GetService("RunService").Heartbeat:Connect(function()  
        if _G.punchanim then  
            local punchTool =  
                player.Backpack:FindFirstChild("Punch") or  
                game.Workspace:FindFirstChild(playerName) and game.Workspace:FindFirstChild(playerName):FindFirstChild("Punch")  

            if punchTool then  
                if punchTool.Parent ~= game.Workspace:FindFirstChild(playerName) then  
                    punchTool.Parent = game.Workspace:FindFirstChild(playerName)
                end  
                punchTool:Activate()
            else  
                warn("Punch tool not found")  
                _G.punchanim = false
            end  
        end  
    end)
end

local function autoRebirth()
    local player = game:GetService("Players").LocalPlayer
    local rebirths = player.leaderstats.Rebirths
    local rebirthRemote = game:GetService("ReplicatedStorage").rEvents.rebirthRemote
    
    while autoRebirthActive and rebirths.Value < targetRebirth do
        rebirthRemote:InvokeServer("rebirthRequest")
        task.wait(0.05)

        if rebirths.Value >= targetRebirth then
            autoRebirthActive = false
            print("[System] Success! Reached target:", targetRebirth)
            break
        end
    end
end

RebirthTab:Toggle({
    Title = "Auto Rebirth To Target",
    Callback = function(state)
        if state then
            if targetRebirth > 0 then
                autoRebirthActive = true
                task.spawn(autoRebirth)
            else
                warn("[System] Set a valid target number first!")
            end
        else
            autoRebirthActive = false
        end
    end
})

local function createRockToggle(title, desc, rockName, durability)
    RockTab:Toggle({

        Desc = desc,

            selectrock = rockName
            getgenv().autoFarm = state

                task.wait()

                    for i, v in pairs(game:GetService("Workspace").machinesFolder:GetDescendants()) do
                        if v.Name == "neededDurability" and v.Value == durability and game.Players.LocalPlayer.Character:FindFirstChild("LeftHand") and game.Players.LocalPlayer.Character:FindFirstChild("RightHand") then
                            firetouchinterest(v.Parent.Rock, game:GetService("Players").LocalPlayer.Character.RightHand, 0)
                            firetouchinterest(v.Parent.Rock, game:GetService("Players").LocalPlayer.Character.RightHand, 1)
                            firetouchinterest(v.Parent.Rock, game:GetService("Players").LocalPlayer.Character.LeftHand, 0)
                            firetouchinterest(v.Parent.Rock, game:GetService("Players").LocalPlayer.Character.LeftHand, 1)
                            for i, v in pairs(game.Players.LocalPlayer.Backpack:GetChildren()) do
                                if v.Name == "Punch" and game.Players.LocalPlayer.Character:FindFirstChild("Humanoid") then
                                    game.Players.LocalPlayer.Character.Humanoid:EquipTool(v)
                                end
                            end
                            game:GetService("Players").LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
                            game:GetService("Players").LocalPlayer.muscleEvent:FireServer("punch", "rightHand")
                        end
                    end
                end
            end
        end
    })
end

-- Function to use machine (teleport once)
local function setupMachine(machineName)
    pcall(function()
        local machine = workspace.machinesFolder:FindFirstChild(machineName)
        if machine and machine:FindFirstChild("interactSeat") then
            local character = game.Players.LocalPlayer.Character
            if character and character:FindFirstChild("HumanoidRootPart") then
                character.HumanoidRootPart.CFrame = machine.interactSeat.CFrame
                task.wait(0.2)
                currentMachine = machineName
                currentMachineActive = true
            end
        end
    end)
end

-- Function to stop using machine
local function stopMachine()
    pcall(function()
        local character = game.Players.LocalPlayer.Character
        if character then
            character.Humanoid.Sit = false
            character.Humanoid.Jump = true
            currentMachine = nil
            currentMachineActive = false
        end
    end)
end

-- Function to do reps on current machine
local function doMachineReps()
    if currentMachineActive and currentMachine then
        pcall(function()
            local machine = workspace.machinesFolder:FindFirstChild(currentMachine)
            if machine and machine:FindFirstChild("interactSeat") then
                game:GetService("ReplicatedStorage").rEvents.machineInteractRemote:InvokeServer("useMachine", machine.interactSeat)
                game:GetService("Players").LocalPlayer.muscleEvent:FireServer("rep", machine.interactSeat)
            end
        end)
    end
end

-- Machine Toggle Creator Function
local function createMachineToggle(parent, title, machineName, stateVar)
    parent:Toggle({
        Title = title,
        Desc = "Auto use " .. machineName,
        Value = false,
        Callback = function(state)
            stateVar = state
            if state then
                stopMachine()
                task.wait(0.5)
                setupMachine(machineName)
                while stateVar and currentMachine == machineName do
                    doMachineReps()
                    task.wait(0.01)
                end
            else
                if currentMachine == machineName then
                    stopMachine()
                end
            end
        end
    })
end

-- Helper function to format numbers
local function formatNumber(value)
    local rawFormatted = tostring(value):reverse():gsub("%d%d%d", "%1,"):reverse():gsub("^,", "")
    
    local abbreviated = ""
    if value >= 1000000000000000000 then
        abbreviated = string.format("%.1fQi", value / 1000000000000000000)
    elseif value >= 1000000000000000 then
        abbreviated = string.format("%.1fQa", value / 1000000000000000)
    elseif value >= 1000000000000 then
        abbreviated = string.format("%.1fT", value / 1000000000000)
    elseif value >= 1000000000 then
        abbreviated = string.format("%.1fB", value / 1000000000)
    elseif value >= 1000000 then
        abbreviated = string.format("%.1fM", value / 1000000)
    elseif value >= 1000 then
        abbreviated = string.format("%.1fK", value / 1000)
    else
        abbreviated = tostring(value)
    end
    
    return rawFormatted, abbreviated
end

-- Helper function to create stat display
local function createStatDisplay(statName, emoji, statPath, isLeaderstat)
    local label = StatsTab:Label({Title = emoji .. " " .. statName, Desc = "0 (0) | GAINED: 0 (0)"})
    local startValue = isLeaderstat and 
        game.Players.LocalPlayer.leaderstats[statPath].Value or 
        game.Players.LocalPlayer[statPath].Value
    
    task.spawn(function()
        while task.wait(1) do
            local currentValue = isLeaderstat and 
                game.Players.LocalPlayer.leaderstats[statPath].Value or 
                game.Players.LocalPlayer[statPath].Value
            local gained = currentValue - startValue
            
            local currentRaw, currentAbbr = formatNumber(currentValue)
            local gainedRaw, gainedAbbr = formatNumber(gained)
            
            label:SetDesc(string.format("%s (%s) | GAINED: %s (%s)", 
                currentRaw, currentAbbr, gainedRaw, gainedAbbr))
        end
    end)
    
    return label
end

-- Function to format numbers with abbreviations
local function formatNumber(num)
    if not num then return "0" end
    num = tonumber(num) or 0
    
    local formats = {
        {1e18, "Qi"}, {1e15, "Qa"}, {1e12, "T"}, 
        {1e9, "B"}, {1e6, "M"}, {1e3, "K"}
    }
    
    for _, format in ipairs(formats) do
        if num >= format[1] then
            return string.format("%.1f%s", num / format[1], format[2])
        end
    end
    return tostring(math.floor(num))
end

-- Function to parse input with abbreviations (Max: 10000Qa)
local function parseNumber(input)
    if not input then return 0 end
    
    input = tostring(input):upper():gsub("%s+", "")
    
    local multipliers = {
        ["QI"] = 1e18, ["QA"] = 1e15,
        ["T"] = 1e12, ["TR"] = 1e12,
        ["B"] = 1e9, ["BN"] = 1e9,
        ["M"] = 1e6, ["ML"] = 1e6,
        ["K"] = 1e3, ["KN"] = 1e3
    }
    
    -- Check if it's just a number
    local plainNum = tonumber(input)
    if plainNum then
        return math.min(plainNum, 10000e15) -- Cap at 10000Qa
    end
    
    -- Check for abbreviation format (e.g., "1.5K", "100M", "2.3B")
    for abbrev, multiplier in pairs(multipliers) do
        local pattern = "^([%d%.]+)" .. abbrev .. "$"
        local match = input:match(pattern)
        if match then
            local num = tonumber(match)
            if num then
                return math.min(num * multiplier, 10000e15) -- Cap at 10000Qa
            end
        end
    end
    
    return 0
end

MiscTab:Section({Title = "Anti AFK"})


MiscTab:Toggle({
    Title = "Anti AFK",
    Desc = "Prevent being kicked for inactivity",
    Value = true, -- Default OFF
    Callback = function(state)
        if state then
            -- Create Fading Display
            local DisplayGui = Instance.new("ScreenGui")
            local DisplayLabel = Instance.new("TextLabel")
            local TimerLabel = Instance.new("TextLabel")
            
            DisplayGui.Name = "AntiAFKDisplay"
            DisplayGui.Parent = game.CoreGui
            DisplayGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
            
            DisplayLabel.Name = "DisplayLabel"
            DisplayLabel.Parent = DisplayGui
            DisplayLabel.BackgroundTransparency = 1
            DisplayLabel.Position = UDim2.new(0.4, 0, 0.01, 0)
            DisplayLabel.Size = UDim2.new(0.2, 0, 0, 30)
            DisplayLabel.Font = Enum.Font.GothamBlack
            DisplayLabel.Text = "APEX PREDATOR X - ANTI AFK"
            DisplayLabel.TextSize = 18
            DisplayLabel.TextTransparency = 0
            DisplayLabel.TextXAlignment = Enum.TextXAlignment.Center
            
            -- Gradient for DisplayLabel
            local displayGradient = Instance.new("UIGradient")
            displayGradient.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(28, 216, 210)),   -- #1CD8D2
                ColorSequenceKeypoint.new(1, Color3.fromRGB(147, 237, 199))   -- #93EDC7
            })
            displayGradient.Rotation = 0
            displayGradient.Parent = DisplayLabel
            
            TimerLabel.Name = "TimerLabel"
            TimerLabel.Parent = DisplayGui
            TimerLabel.BackgroundTransparency = 1
            TimerLabel.Position = UDim2.new(0.4, 0, 0.045, 0)
            TimerLabel.Size = UDim2.new(0.2, 0, 0, 30)
            TimerLabel.Font = Enum.Font.GothamBold
            TimerLabel.Text = "00:00:00"
            TimerLabel.TextSize = 12
            TimerLabel.TextTransparency = 0
            TimerLabel.TextXAlignment = Enum.TextXAlignment.Center
            
            -- Gradient for TimerLabel
            local timerGradient = Instance.new("UIGradient")
            timerGradient.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(28, 216, 210)),   -- #1CD8D2
                ColorSequenceKeypoint.new(1, Color3.fromRGB(147, 237, 199))   -- #93EDC7
            })
            timerGradient.Rotation = 0
            timerGradient.Parent = TimerLabel
            
            -- Animate gradients
            _G.animateGradients = true
            task.spawn(function()
                while _G.animateGradients do
                    for i = 0, 360, 2 do
                        if not _G.animateGradients then break end
                        displayGradient.Rotation = i
                        timerGradient.Rotation = i
                        task.wait(0.03)
                    end
                end
            end)
            
            -- Animation loop for fading
            _G.displayAFK = true
            task.spawn(function()
                while _G.displayAFK do
                    -- Fade out both
                    for i = 0, 1, 0.1 do
                        if not _G.displayAFK then break end
                        DisplayLabel.TextTransparency = i
                        TimerLabel.TextTransparency = i
                        task.wait(0.05)
                    end
                    
                    if not _G.displayAFK then break end
                    task.wait(0.5)
                    
                    -- Fade in both
                    for i = 1, 0, -0.1 do
                        if not _G.displayAFK then break end
                        DisplayLabel.TextTransparency = i
                        TimerLabel.TextTransparency = i
                        task.wait(0.05)
                    end
                    
                    if not _G.displayAFK then break end
                    task.wait(3)
                end
            end)
            
            -- Timer
            local seconds, minutes, hours = 0, 0, 0
            _G.timerAFK = true
            task.spawn(function()
                while _G.timerAFK do
                    seconds = seconds + 1
                    task.wait(1)
                    if seconds >= 60 then seconds = 0; minutes = minutes + 1 end
                    if minutes >= 60 then minutes = 0; hours = hours + 1 end
                    TimerLabel.Text = string.format("%02d:%02d:%02d", hours, minutes, seconds)
                end
            end)
            
            -- Anti-AFK function
            local vu = game:GetService('VirtualUser')
            game.Players.LocalPlayer.Idled:Connect(function()
                vu:CaptureController()
                vu:ClickButton2(Vector2.new())
            end)
            
            Window:Notify({
                Title = "Anti AFK",
                Desc = "Anti AFK activated with animated timer!",
                Time = 3
            })
        else
            -- Turn off
            _G.displayAFK = false
            _G.timerAFK = false
            _G.animateGradients = false
            if game.CoreGui:FindFirstChild("AntiAFKDisplay") then
                game.CoreGui.AntiAFKDisplay:Destroy()
            end
            Window:Notify({
                Title = "Anti AFK",
                Desc = "Anti AFK deactivated!",
                Time = 3
            })
        end
    end
})
